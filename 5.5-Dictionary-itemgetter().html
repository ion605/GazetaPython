Accesarea valorilor dintr-un dicționar - Accessing Dictionary Values

operator.itemgetter construiește un obiect apelabil reutilizabil care extrage valori dintr-un dicționar pe baza cheilor acestora. Această abordare este adesea mai eficientă și mai elegantă decât scrierea unor bucle explicite sau a unor construcții de tip list comprehension", în special atunci când se extrag aceleași chei din mai multe dicționare.

operator.itemgetter constructs a reusable callable that fetches values from a dictionary by their keys. This approach is often more efficient and elegant than writing explicit loops or list comprehensions, especially when extracting the same keys from multiple dictionaries.

# Accessing a Single Value
from operator import itemgetter

membru = {"nume": "Costel", "assoc": "#4", "rol": "Administrator"}

# Creați instanțe getter pentru cheile 'nume' si 'rol'
numele = itemgetter("nume")
admin = itemgetter("rol")

# Apelați-le prin transmiterea dicționarului
print(f"Numele {admin(membru)}ului: {numele(membru)}")
==> Numele Administratorului: Costel

# Accesarea simultană a mai multor valori
from operator import itemgetter

membru = {"nume": "Costel", "assoc": "#4", "rol": "Administrator"}

# Creează o funcție de tip getter pentru mai multe chei
profil = itemgetter("nume", "rol")


# Apelați-o prin transmiterea dicționarului
print(f"Numele si rolul: {profil(membru)}")
==> Numele si rolul: ('Costel', 'Administrator')

# Extracting Elements Across a List of Dictionaries
from operator import itemgetter

membri = [
    {"nume": "Cuta", "rol": "Presedinte"},
    {"nume": "Pupu", "rol": "Secretar"},
    {"nume": "Tucu", "rol": "Administrator"},
]

# Extrage toate rolurile într-o singură listă
get_rol = itemgetter("rol")
roluri = list(map(get_rol, membri))
print(roluri) ==> ['Presedinte', 'Secretar', 'Administrator']

# Lista de [{elev: [note]}] direct din dicționarul principal
from operator import itemgetter

scor_elevi = {"scoruri": [{"virgil": [88, 76, 92, 100]}, {"tatiana": [88, 76, 92, 85]}, {"sever": [88, 76, 92, 85]}]}

# Creează funcția de extragere pentru cheia "scoruri"
obtine_scoruri = itemgetter("scoruri")

elevi_note = obtine_scoruri(scor_elevi)
print(elevi_note) ==> [{'virgil': [88, 76, 92, 100]}, {'tatiana': [88, 76, 92, 85]}, {'sever': [88, 76, 92, 85]}]

# Notele unui anumit elev
from operator import itemgetter

scor_elevi = {"scoruri": [{"virgil": [88, 76, 92, 100]}, {"tatiana": [88, 76, 92, 85]}, {"sever": [88, 76, 92, 85]}]}
obtine_scoruri = itemgetter("scoruri")
elevi_note = obtine_scoruri(scor_elevi)
print(elevi_note) ==> [{'virgil': [88, 76, 92, 100]}, {'tatiana': [88, 76, 92, 85]}, {'sever': [88, 76, 92, 85]}]

obtine_virgil = itemgetter("virgil")
print(obtine_virgil) ==> operator.itemgetter('virgil')

note_virgil = obtine_virgil(elevi_note[0])
print(f"Notele lui Virgil: {note_virgil}") ==> Notele lui Virgil: [88, 76, 92, 100]

Dicționare, dintre care unele ar putea să nu conțină o anumită cheie.

The dictionaries, some of which might not contain a specific key.

from operator import itemgetter

# Un dicționar cu câmpuri standard, dar căruia îi lipsește cheia "email"
membru = {"nume": "Costel", "assoc": "#4", "rol": "Administrator"}

# Definirea unui itemgetter care caută "utilizator" și "email"
obtine_info_contact = itemgetter("utilizator", "email")

try:
    # Această operațiune va eșua și va genera o eroare de tip KeyError, deoarece "email" nu există
    contact_info = obtine_info_contact(membru)
    print("Succes:", contact_info)

except KeyError as error:
    # Numele cheii lipsă este atașat argumentului erorii
    print(f"Eroare: Cheia {error} nu există în dicționar.")
==> Eroare: Cheia 'utilizator' nu există în dicționar.

Excepția poate fi gestionată direct în cod pentru a furniza valori implicite.

The exception can be handled directly in the code to provide default values.

from operator import itemgetter

membrii = [
    {"nume": "Costel", "assoc": "#4", "rol": "Administrator"},
    {"nume": "Vecin", "assoc": "#4"},  # Lipsește "rol"
]

obtine_rol = itemgetter("rol")

for membru in membrii:
    try:
        rol = obtine_rol(membru)
    except KeyError:
        rol = "Musafir"  # Valoare implicită de rezervă

    print(f"{membru['nume']}: {rol}")
==> Costel: Administrator
==> Vecin: Musafir

Sortareaa valorilor dintr-un dicționar - Sorting Dictionary Values

# Criteriu: Sortare după media notelor (Descrescător)
from operator import itemgetter

scor_elevi = {"scoruri": [{"virgil": [88, 76, 82, 80]}, {"tatiana": [82, 70, 75, 85]}, {"sever": [88, 76, 92, 85]}]}
lista_elevi = scor_elevi["scoruri"]

def dupa_medie(element_lista):
    # element_lista este un dicționar precum {"virgil": [...]}
    nume = list(element_lista.keys())[0]
    note = element_lista[nume]
    return sum(note) / len(note)

# Sortăm lista originală
elevi_sortati = sorted(lista_elevi, key=dupa_medie, reverse=True)

print(elevi_sortati)
==> [{'sever': [88, 76, 92, 85]}, {'virgil': [88, 76, 82, 80]}, {'tatiana': [82, 70, 75, 85]}]

# O structură mai eficientă
import operator

elevi_optimizat = [
    {"nume": "virgil", "medie": 81.5},
    {"nume": "tatiana", "medie": 78.0},
    {"nume": "sever","medie": 85.25}
]

# Sortare alfabetică după nume folosind itemgetter
elevi_alfabetic = sorted(elevi_optimizat, key=operator.itemgetter("nume"))
print(elevi_alfabetic)
==> [{'nume': 'sever', 'medie': 85.25}, {'nume': 'tatiana', 'medie': 78.0}, {'nume': 'virgil', 'medie': 81.5}]

# Sortare după medie (descrescător) folosind itemgetter
elevi_dupa_medie = sorted(elevi_optimizat, key=operator.itemgetter("medie"), reverse=True)
print(elevi_dupa_medie)
==> [{'nume': 'sever', 'medie': 85.25}, {'nume': 'virgil', 'medie': 81.5}, {'nume': 'tatiana', 'medie': 78.0}]

Pentru operațiuni complexe de sortare, este adesea mai simplu să folosești o funcție lambda sau să schimbi structura dicționarului într-una mai plată.

For complex sorting operations, it is often simpler to use a lambda function or to change the dictionary structure into a flatter one.

Performanță - Performance

Performanța sortării dicționar vs listă - Performance sorting: dictionary vs. list

from timeit import timeit
dict_articole_dict = {
    'articol1': {'nume': 'caise', 'cantitate': 10, 'cost': 2.40},
    'articol4': {'nume': 'gutui', 'cantitate': 15, 'cost': 2.12},
    'articol3': {'nume': 'mere', 'cantitate': 50, 'cost': 1.69},
    'articol2': {'nume': 'pere', 'cantitate': 25, 'cost': 1.5},
}

lista_articole_dict = [
    {'articol':1, 'nume': 'caise', 'cantitate': 10, 'cost': 2.40},
    {'articol':4, 'nume': 'gutui', 'cantitate': 15, 'cost': 2.12},
    {'articol':3, 'nume': 'mere', 'cantitate': 50, 'cost': 1.69},
    {'articol':2, 'nume': 'pere', 'cantitate': 25, 'cost': 1.5}
]

sortarea_listei = "sorted(lista_articole_dict, key=lambda item:item['cost'])"
sortarea_dict = """
dict(sorted(dict_articole_dict.items(), key=lambda item: item[1]['cost']))
"""
sortare_lista_time = timeit(stmt=sortarea_listei, globals=globals())
sortare_dict_time = timeit(stmt=sortarea_dict, globals=globals())

print(
    f"""\
{sortare_lista_time=:.2f} secunde
{sortare_dict_time=:.2f} secunde
lista este de {(sortare_dict_time/sortare_lista_time):.2f} mai rapida"""
)
==> sortare_lista_time=0.79 secunde
==> sortare_dict_time=1.51 secunde
==> lista este de 1.90 mai rapida

Performanța căutărilor cu for - Performance of lookups with for

# compararea sortării dicționar vs listă
from timeit import timeit


dict_articole_dict = {
    1: {'nume': 'caise', 'cantitate': 10, 'cost': 2.40},
    4: {'nume': 'gutui', 'cantitate': 15, 'cost': 2.12},
    3: {'nume': 'mere', 'cantitate': 50, 'cost': 1.69},
    2: {'nume': 'pere', 'cantitate': 25, 'cost': 1.5},
}

lista_articole_dict = [
    {'id':1, 'nume': 'caise', 'cantitate': 10, 'cost': 2.40},
    {'id':4, 'nume': 'gutui', 'cantitate': 15, 'cost': 2.12},
    {'id':3, 'nume': 'mere', 'cantitate': 50, 'cost': 1.69},
    {'id':2, 'nume': 'pere', 'cantitate': 25, 'cost': 1.5}
]


cautari = [3, 2, 4, 1, 2, 2]

configurare_lista = """
def cheie_din_lista(key):
    for item in lista_articole_dict:
        if item["id"] == key:
            return item
"""

lista_de_căutare = """
for key in cautari:
    cheie_din_lista(key)
"""

dict_de_căutare = """
for key in cautari:
    dict_articole_dict[key]
"""

timp_lista_cautare = timeit(stmt=lista_de_căutare, setup=configurare_lista, globals=globals())
timp_dict_cautare = timeit(stmt=dict_de_căutare, globals=globals())

print(
    f"""\
{timp_lista_cautare=:.2f} secunde
{timp_dict_cautare=:.2f} secunde
dict este de {(timp_lista_cautare / timp_dict_cautare):.2f} ori mai rapid"""
)
==> timp_lista_cautare=2.16 secunde
==> timp_dict_cautare=0.41 secunde
==> dict este de 5.29 ori mai rapid

Performanță folosind itemgetter() - Performance using itemgetter()

Utilizarea operator.itemgetter() în Python este, în general, mai rapidă și mai elegantă decât folosirea funcțiilor lambda sau a buclelor atunci când se efectuează sortări.

Using operator.itemgetter() in Python is generally faster and more elegant than using lambda functions or loops when performing sorts.

from operator import itemgetter
from timeit import timeit

dict_to_order = {1: "rug", 2: "par", 3: "jug", 4: "infiintat", 5: "urs", 6: "albastru",7: "negru", 8: "perna",
                 9: "trece", 10: "jos13", 11: "nucleul", 12: "biblioteca22", 13: "p2transfer", 14: "sase", 15: "informatii",
                 16: "opus-ziua", 17: "index", 18: "certificat", 19: "extensii-tipuri", 20: "normalizator", 21: "zare-aurie"
}

sortat_cu_lambda = "sorted(dict_to_order.items(), key=lambda item: item[1])"
sortat_cu_itemgetter = "sorted(dict_to_order.items(), key=itemgetter(1))"

sortat_cu_lambda_timp = timeit(stmt=sortat_cu_lambda, globals=globals())
sortat_cu_itemgetter_timp = timeit(stmt=sortat_cu_itemgetter, globals=globals())

print(f"""\
{sortat_cu_lambda_timp=:.2f} secunde
{sortat_cu_itemgetter_timp=:.2f} secunde
itemgetter is {(sortat_cu_lambda_timp / sortat_cu_itemgetter_timp):.2f} mai rapid""")
==> sortat_cu_lambda_timp=3.78 secunde
==> sortat_cu_itemgetter_timp=2.52 secunde
==> itemgetter is 1.50 mai rapid
